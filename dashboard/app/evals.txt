"use client";

import { useEffect, useMemo, useState } from "react";
import {
  AlertTriangle,
  BarChart3,
  CheckCircle2,
  RefreshCw,
  XCircle,
} from "lucide-react";
import {
  Bar,
  BarChart,
  CartesianGrid,
  ResponsiveContainer,
  Tooltip,
  XAxis,
  YAxis,
} from "recharts";

type EvalResult = {
  id?: number;
  q?: string;
  question?: string;
  expected?: string;
  actual?: string;
  correct?: boolean;
  citation_correct?: boolean;
  escalated?: boolean;
  expected_escalated?: boolean;
  latency_ms?: number;
};

type EvalFile = {
  correctness?: number;
  citation_accuracy?: number;
  escalation_accuracy?: number;
  avg_latency_ms?: number;
  token_cost?: number | null;
  results?: EvalResult[];
};

const API_BASE = "http://127.0.0.1:8000";

export default function EvalsPage() {
  const [data, setData] = useState<EvalFile | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  useEffect(() => {
    loadEvals();
  }, []);

  async function loadEvals() {
    setLoading(true);
    setError("");

    try {
      const response = await fetch(`${API_BASE}/evals`);

      if (!response.ok) {
        throw new Error("Failed to load evaluation results.");
      }

      const result = await response.json();
      setData(result);
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Could not connect to the backend."
      );
    } finally {
      setLoading(false);
    }
  }

  const chartData = useMemo(() => {
    if (!data) {
      return [];
    }

    return [
      {
        name: "Correctness",
        score: Math.round((data.correctness || 0) * 100),
      },
      {
        name: "Citation Accuracy",
        score: Math.round((data.citation_accuracy || 0) * 100),
      },
      {
        name: "Escalation Accuracy",
        score: Math.round((data.escalation_accuracy || 0) * 100),
      },
    ];
  }, [data]);

  const failedCases = useMemo(() => {
    if (!data?.results) {
      return [];
    }

    return data.results.filter(
      (item) =>
        item.correct === false ||
        item.citation_correct === false ||
        item.escalated !== item.expected_escalated
    );
  }, [data]);

  return (
    <main className="min-h-screen bg-[#f8f7f5] text-[#252326]">
      <div className="mx-auto max-w-7xl px-6 py-8 lg:px-10">
        <header className="flex items-center justify-between border-b border-[#e8e3e1] pb-6">
          <div>
            <p className="text-sm font-medium uppercase tracking-[0.2em] text-[#a46b78]">
              SupportOps
            </p>

            <h1 className="mt-1 text-2xl font-semibold tracking-tight">
              Evaluations
            </h1>

            <p className="mt-2 text-sm text-[#716a6e]">
              Measure answer quality, citations, escalation behavior, and
              latency.
            </p>
          </div>

          <button
            onClick={loadEvals}
            className="flex items-center gap-2 rounded-xl border border-[#e8e3e1] bg-white px-4 py-2.5 text-sm font-medium text-[#514b4f] shadow-sm transition hover:bg-[#faf8f8]"
          >
            <RefreshCw size={16} />
            Refresh
          </button>
        </header>

        <section className="py-8">
          {loading && (
            <div className="rounded-2xl border border-[#e8e3e1] bg-white p-8 text-center text-[#716a6e]">
              Loading evaluation results...
            </div>
          )}

          {!loading && error && (
            <div className="rounded-2xl border border-red-200 bg-white p-6">
              <div className="flex items-start gap-3">
                <AlertTriangle
                  size={20}
                  className="mt-0.5 text-red-500"
                />

                <div>
                  <h2 className="font-semibold">
                    Could not load evaluations
                  </h2>

                  <p className="mt-2 text-sm text-[#716a6e]">
                    {error}
                  </p>

                  <p className="mt-3 text-sm text-[#716a6e]">
                    The FastAPI endpoint for evaluations may not exist yet.
                  </p>
                </div>
              </div>
            </div>
          )}

          {!loading && !error && data && (
            <>
              <section className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4">
                <MetricCard
                  icon={<CheckCircle2 size={20} />}
                  label="Correctness"
                  value={`${((data.correctness || 0) * 100).toFixed(1)}%`}
                />

                <MetricCard
                  icon={<BarChart3 size={20} />}
                  label="Citation Accuracy"
                  value={`${(
                    (data.citation_accuracy || 0) * 100
                  ).toFixed(1)}%`}
                />

                <MetricCard
                  icon={<AlertTriangle size={20} />}
                  label="Escalation Accuracy"
                  value={`${(
                    (data.escalation_accuracy || 0) * 100
                  ).toFixed(1)}%`}
                />

                <MetricCard
                  icon={<BarChart3 size={20} />}
                  label="Average Latency"
                  value={`${Math.round(data.avg_latency_ms || 0)} ms`}
                />
              </section>

              <section className="mt-6 rounded-2xl border border-[#e8e3e1] bg-white p-6 shadow-sm">
                <div>
                  <h2 className="text-lg font-semibold">
                    Evaluation Metrics
                  </h2>

                  <p className="mt-1 text-sm text-[#716a6e]">
                    Latest evaluation scores across the test set.
                  </p>
                </div>

                <div className="mt-6 h-[320px] w-full">
                  <ResponsiveContainer width="100%" height="100%">
                    <BarChart data={chartData}>
                      <CartesianGrid strokeDasharray="3 3" />
                      <XAxis dataKey="name" />
                      <YAxis domain={[0, 100]} />
                      <Tooltip />
                      <Bar dataKey="score" radius={[8, 8, 0, 0]} />
                    </BarChart>
                  </ResponsiveContainer>
                </div>
              </section>

              <section className="mt-6 rounded-2xl border border-[#e8e3e1] bg-white shadow-sm">
                <div className="flex items-center justify-between border-b border-[#eee9e7] px-6 py-5">
                  <div>
                    <h2 className="text-lg font-semibold">
                      Failed Evaluation Cases
                    </h2>

                    <p className="mt-1 text-sm text-[#716a6e]">
                      Cases where correctness, citations, or escalation
                      behavior did not match expectations.
                    </p>
                  </div>

                  <span className="rounded-full bg-red-50 px-3 py-1.5 text-xs font-medium text-red-700">
                    {failedCases.length} failed
                  </span>
                </div>

                {failedCases.length === 0 ? (
                  <div className="p-10 text-center">
                    <CheckCircle2
                      size={28}
                      className="mx-auto text-emerald-500"
                    />

                    <p className="mt-3 font-medium">
                      No failed cases
                    </p>

                    <p className="mt-1 text-sm text-[#716a6e]">
                      All evaluation cases passed.
                    </p>
                  </div>
                ) : (
                  <div className="divide-y divide-[#eee9e7]">
                    {failedCases.map((item, index) => (
                      <div
                        key={item.id ?? index}
                        className="p-6"
                      >
                        <div className="flex items-start justify-between gap-4">
                          <div className="flex-1">
                            <div className="flex items-center gap-2">
                              <XCircle
                                size={18}
                                className="text-red-500"
                              />

                              <span className="text-xs font-semibold uppercase tracking-wider text-[#a46b78]">
                                Case {item.id ?? index + 1}
                              </span>
                            </div>

                            <p className="mt-3 text-sm font-medium leading-6">
                              {item.q || item.question || "Unknown question"}
                            </p>
                          </div>
                        </div>

                        <div className="mt-4 grid gap-4 md:grid-cols-3">
                          <InfoBox
                            label="Expected"
                            value={item.expected || "Not available"}
                          />

                          <InfoBox
                            label="Actual"
                            value={item.actual || "Not available"}
                          />

                          <InfoBox
                            label="Escalation"
                            value={
                              item.escalated
                                ? "Escalated"
                                : "Not escalated"
                            }
                          />
                        </div>
                      </div>
                    ))}
                  </div>
                )}
              </section>
            </>
          )}
        </section>
      </div>
    </main>
  );
}

function MetricCard({
  icon,
  label,
  value,
}: {
  icon: React.ReactNode;
  label: string;
  value: string;
}) {
  return (
    <div className="rounded-2xl border border-[#e8e3e1] bg-white p-5 shadow-sm">
      <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-[#f3e3e7] text-[#a46b78]">
        {icon}
      </div>

      <p className="mt-5 text-sm text-[#81797e]">
        {label}
      </p>

      <p className="mt-1 text-2xl font-semibold tracking-tight">
        {value}
      </p>
    </div>
  );
}

function InfoBox({
  label,
  value,
}: {
  label: string;
  value: string;
}) {
  return (
    <div className="rounded-xl border border-[#e8e3e1] bg-[#faf8f8] p-4">
      <p className="text-xs font-semibold uppercase tracking-wider text-[#81797e]">
        {label}
      </p>

      <p className="mt-2 whitespace-pre-wrap text-sm leading-6 text-[#514b4f]">
        {value}
      </p>
    </div>
  );
}
